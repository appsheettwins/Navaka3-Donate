โค้ดสคริปต์ที่ต้องนำไปวางใน Google Apps Script (ปรับใหม่แล้ว)

💡 โค้ดนี้รองรับทั้งแบบ **ต่อท้ายข้อมูล (Append)** และแบบ **เขียนทับ (Overwrite)** กรุณานำโค้ดนี้ไปวางแทนอันเดิมใน Apps Script แล้วกด Deploy ใหม่

  1. เปิดไฟล์ Google Sheet ID: 1ppxAhozZszl8_x8qRSy9EKTDhsgrx5RE8QW23k5CYTI
  2. ไปที่เมนู ส่วนขยาย (Extensions) > Apps Script
  3. วางโค้ดชุดนี้แทนที่ไฟล์เดิม แล้วกด บันทึก (Save)
  4. กดปุ่ม ทำให้ใช้งานได้ (Deploy) > การทำให้ใช้งานได้ใหม่ (New deployment)
  5. เลือกประเภท เว็บแอป (Web App), สิทธิ์เข้าถึงเป็น ทุกคน (Anyone) แล้วกด Deploy
function doPost(e) {
  try {
    var ss = SpreadsheetApp.openById('1ppxAhozZszl8_x8qRSy9EKTDhsgrx5RE8QW23k5CYTI');
    var sheet = ss.getSheets()[0];
    var data = JSON.parse(e.postData.contents);
    
    // Header แถวที่ 1
    var headers = ['ลำดับ', 'ชื่อผู้ถวาย/ร่วมบุญ', 'จำนวนเงิน', 'ตัวอักษร', 'การชำระ', 'ผู้บอกบุญ', 'วันที่ทำบุญ'];

    // กรณีที่ 0: บันทึกและซิงก์ URL ใหม่ไปยังแผ่นงาน Config (สำหรับอัปเดตทุกเครื่อง)
    if (data.action === 'save_url' && data.url) {
      var configSheet = ss.getSheetByName('Config') || ss.insertSheet('Config');
      configSheet.getRange('A1').setValue(data.url);
      return ContentService.createTextOutput(JSON.stringify({status: 'success', message: 'บันทึก URL ซิงก์ทุกเครื่องสำเร็จ'})).setMimeType(ContentService.MimeType.JSON);
    }

    // กรณีที่ 1: บันทึกเฉพาะข้อมูลใหม่ต่อท้าย (Append Only) ไม่ลบอันเดิม
    if (data.action === 'append') {
      if (sheet.getLastRow() === 0) {
        sheet.appendRow(headers);
      }
      data.donors.forEach(function(d) {
        sheet.appendRow([d.serial, d.name, d.amountNum, d.amountText, d.payType, d.referrer, d.donationDate]);
      });
      return ContentService.createTextOutput(JSON.stringify({status: 'success', message: 'ต่อท้ายข้อมูลสำเร็จ'})).setMimeType(ContentService.MimeType.JSON);
    }
    
    // กรณีที่ 2: เขียนทับข้อมูลทั้งหมด (Full Sync)
    if (data.action === 'save') {
      sheet.clearContents();
      sheet.appendRow(headers);
      data.donors.forEach(function(d) {
        sheet.appendRow([d.serial, d.name, d.amountNum, d.amountText, d.payType, d.referrer, d.donationDate]);
      });
      return ContentService.createTextOutput(JSON.stringify({status: 'success', message: 'บันทึกทับทั้งหมดสำเร็จ'})).setMimeType(ContentService.MimeType.JSON);
    }
  } catch(err) {
    return ContentService.createTextOutput(JSON.stringify({status: 'error', message: err.toString()})).setMimeType(ContentService.MimeType.JSON);
  }
}

function doGet(e) {
  try {
    var ss = SpreadsheetApp.openById('1ppxAhozZszl8_x8qRSy9EKTDhsgrx5RE8QW23k5CYTI');
    var sheet = ss.getSheets()[0];
    var rows = sheet.getDataRange().getDisplayValues();
    var donors = [];
    for (var i = 1; i < rows.length; i++) {
      if (rows[i][1] || rows[i][0]) {
        donors.push({
          serial: rows[i][0] || '',
          name: rows[i][1] || '',
          amountNum: parseFloat(rows[i][2].toString().replace(/,/g, '')) || 0,
          amountText: rows[i][3] || '',
          payType: rows[i][4] || 'เงินโอน',
          referrer: rows[i][5] || '',
          donationDate: rows[i][6] || ''
        });
      }
    }

    // ดึง URL ล่าสุดจากแผ่นงาน Config
    var currentUrl = '';
    var configSheet = ss.getSheetByName('Config');
    if (configSheet) {
      currentUrl = configSheet.getRange('A1').getValue() || '';
    }

    return ContentService.createTextOutput(JSON.stringify({
      status: 'success', 
      donors: donors,
      currentUrl: currentUrl
    })).setMimeType(ContentService.MimeType.JSON);
  } catch(err) {
    return ContentService.createTextOutput(JSON.stringify({status: 'error', message: err.toString()})).setMimeType(ContentService.MimeType.JSON);
  }
}
