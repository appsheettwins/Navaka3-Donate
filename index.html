วิธีเชื่อมต่อและสร้างสคริปต์ Google Apps Script (คลิกเปิด)
  1. เปิดไฟล์ Google Sheet ID: 1ppxAhozZszl8_x8qRSy9EKTDhsgrx5RE8QW23k5CYTI
  2. ไปที่เมนู ส่วนขยาย (Extensions) > Apps Script
  3. วางโค้ดชุดนี้แทนที่ไฟล์เดิม แล้วกด บันทึก (Save)
  4. กดปุ่ม ทำให้ใช้งานได้ (Deploy) > การทำให้ใช้งานได้ใหม่ (New deployment)
  5. เลือกประเภท เว็บแอป (Web App), สิทธิ์เข้าถึงเป็น ทุกคน (Anyone) แล้วกด Deploy
  6. คัดลอก **URL เว็บแอป** มาวางในช่องด้านบนแล้วกด บันทึก URL
function doPost(e) {
  try {
    var sheet = SpreadsheetApp.openById('1ppxAhozZszl8_x8qRSy9EKTDhsgrx5RE8QW23k5CYTI').getActiveSheet();
    var data = JSON.parse(e.postData.contents);
    if (data.action === 'save') {
      sheet.clearContents();
      sheet.appendRow(['ลำดับ', 'ชื่อผู้ถวาย/ร่วมบุญ', 'จำนวนเงิน', 'ตัวอักษร', 'การชำระ', 'ผู้บอกบุญ', 'วันที่ทำบุญ']);
      data.donors.forEach(function(d) {
        sheet.appendRow([d.serial, d.name, d.amountNum, d.amountText, d.payType, d.referrer, d.donationDate]);
      });
      return ContentService.createTextOutput(JSON.stringify({status: 'success'})).setMimeType(ContentService.MimeType.JSON);
    }
  } catch(err) {
    return ContentService.createTextOutput(JSON.stringify({status: 'error', message: err.toString()})).setMimeType(ContentService.MimeType.JSON);
  }
}

function doGet(e) {
  try {
    var sheet = SpreadsheetApp.openById('1ppxAhozZszl8_x8qRSy9EKTDhsgrx5RE8QW23k5CYTI').getActiveSheet();
    var rows = sheet.getDataRange().getValues();
    var donors = [];
    for (var i = 1; i < rows.length; i++) {
      donors.push({
        serial: rows[i][0] || '',
        name: rows[i][1] || '',
        amountNum: parseFloat(rows[i][2]) || 0,
        amountText: rows[i][3] || '',
        payType: rows[i][4] || 'เงินโอน',
        referrer: rows[i][5] || '',
        donationDate: rows[i][6] || ''
      });
    }
    return ContentService.createTextOutput(JSON.stringify({status: 'success', donors: donors})).setMimeType(ContentService.MimeType.JSON);
  } catch(err) {
    return ContentService.createTextOutput(JSON.stringify({status: 'error', message: err.toString()})).setMimeType(ContentService.MimeType.JSON);
  }
}
